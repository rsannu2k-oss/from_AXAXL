/* Create 2 tables to hold current and earliest Rate of Exchanges per Company into Corporate Base (USD). */

/* 1. Current Rate of Exchange */
SELECT 
	FKCompany,
	ExchangeRate
INTO
	#InfCurrentRoe

FROM

	HistoricalRoe 

WHERE
	FKRateGroup = '*CURRENT'	AND
	FKCcy = 'USD'

/* 2. Earliest Rate of Exchange */
SELECT
	FKCompany,
	ExchangeRate
INTO
	#InfEarliestRoe


FROM

	HistoricalRoe

WHERE
	FKRateGroup LIKE 'X199901%' AND
	FKCcy = 'USD'

/* 3. Specific Period Rate of Exchange */
SELECT 

	FKCompany,
	ExchangeRate,
	RateGroup=LEFT(FKRateGroup,7)
INTO
	#InfRptPeriodDateRoe

FROM

	HistoricalRoe

WHERE
	FKRateGroup LIKE 'X%' AND
	FKCcy='USD'

CREATE UNIQUE CLUSTERED INDEX IX_#InfRptPeriodDateRoe_Key ON #InfRptPeriodDateRoe (FKCompany,RateGroup)


/***	Michael Dunne. 11th May 2004. Remove Lead Cedent Temporary Tables	***/

	As part of the GENIUS Incremental Project split AllTransFullDetail into Gross/Ri & 
	Claims/Non-Claims
	- Hence for this build amend AllTransFullDetail to TransPolicyOutwards and remove the filter	***/

	TransPolicyOutwards.FKGenIntStatsHeader,
	TransPolicyOutwards.FKGenIntStatsSec,
        TransPolicyOutwards.FKGenIntStatsAmt,

	SUM(TransPolicyOutwards.OrigCcyAmt)				AS	SumOrigCcyAmt,
	SUM(TransPolicyOutwards.AcctingCcyAmt)			        AS	SumAcctingCcyAmt,
	SUM(TransPolicyOutwards.BaseCcyAmtRoeDateBooked)		AS	SumBaseCcyAmtRoeDateBooked,
	TransPolicyOutwards.PFKRiPolNbr,
	TransPolicyOutwards.FKGenRiIntPolNbr,
	TransPolicyOutwards.FKGenRiIntPolSeq,
	TransPolicyOutwards.FKGenRiIntSecSeq,
	TransPolicyOutwards.FKGenIntStatsPostingKey,
	TransPolicyOutwards.R_TR_AMT_PAR





INTO	#InfRi

FROM	TransPolicyOutwards

/***	Michael Dunne. 9th August 2006. Filter no longer required as only Ri non-Claim records 
	in TransPolicyOutwards
WHERE
	(AllTransFullDetail.PFKRiPolNbr <> '' AND AllTransFullDetail.PFKRiPolNbr IS NOT NULL)		AND
	AllTransFullDetail.TransColumnNbr <> 5	AND
	AllTransFullDetail.PFKClmNbr = ''
***/


GROUP BY
	TransPolicyOutwards.FKGenIntStatsHeader,
	TransPolicyOutwards.FKGenIntStatsSec,
        TransPolicyOutwards.FKGenIntStatsAmt,
	TransPolicyOutwards.FKRevenueCode,
	TransPolicyOutwards.PFKRiPolNbr,
	TransPolicyOutwards.FKGenRiIntPolNbr,
	TransPolicyOutwards.FKGenRiIntPolSeq,
	TransPolicyOutwards.FKGenRiIntSecSeq,
	TransPolicyOutwards.FKGenIntStatsPostingKey,
	TransPolicyOutwards.R_TR_AMT_PAR

/**	- This is to allow joining of Inwards and Outwards correctly.									***/
HAVING 
	SUM(TransPolicyOutwards.BaseCcyAmtRoeDateBooked) <= -0.001	OR
	SUM(TransPolicyOutwards.BaseCcyAmtRoeDateBooked) >=  0.001

CREATE INDEX #InfRi_PrimaryKey ON #InfRi ([FKGenIntStatsHeader], [FKGenIntStatsSec], [FKRevenueCode])

/*** Total Ri Amounts by Stats Header/Section/Revenue Code 	***/

	TransPolicyOutwards.FKRevenueCode

CREATE INDEX #InfRi_Total_Initial_PrimaryKey ON #InfRi_Total_Initial ([FKGenIntStatsHeader], [FKGenIntStatsSec], [FKRevenueCode])


/* Number of Unique Ri Policies attached at Header, Section and  Revenue Code.		*/

/* This is required in order to calculate a correct ratio in those cases where there are 		*/
/* multiple Ri Policies attached.								*/

SELECT

	#InfRi.FKGenIntStatsHeader,
	#InfRi.FKGenIntStatsSec,
        #InfRi.FKGenIntStatsAmt,
	#InfRi.FKRevenueCode,
	CAST(COUNT(DISTINCT #InfRi.PFKRiPolNbr) AS INTEGER)		AS	NbrRiPolicies

INTO	#InfRi_Total

FROM	#InfRi

GROUP BY 

	#InfRi.FKGenIntStatsHeader,
	#InfRi.FKGenIntStatsSec,
        #InfRi.FKGenIntStatsAmt,
	#InfRi.FKRevenueCode

CREATE INDEX #InfRi_Total_PrimaryKey ON #InfRi_Total ([FKGenIntStatsHeader], [FKGenIntStatsSec], [FKRevenueCode])

/* Ri Percentage for each record */
SELECT
	#InfRi.FKGenIntStatsHeader,
	#InfRi.FKGenIntStatsSec,
        #InfRi.FKGenIntStatsAmt,
	#InfRi.FKRevenueCode,
	#InfRi.FKGenIntStatsPostingKey,
	#InfRi.R_TR_AMT_PAR,
	#InfRi.PFKRiPolNbr,
	#InfRi.FKGenRiIntPolNbr,
	#InfRi.FKGenRiIntPolSeq,
	#InfRi.FKGenRiIntSecSeq,

	CASE
		WHEN	(#InfRi_Total_Initial.SumBaseCcyAmtRoeDateBooked IS NULL) 	THEN 1

	/***	Michael Dunne. 18th April 2005. SQL can calculate zero as only being very nearly zero and hence we cannot check precisely zero here.	***/
	/***
		WHEN	(#InfRi_Total_Initial.SumBaseCcyAmtRoeDateBooked = 0.0) 		THEN (1.00/#InfRi_Total.NbrRiPolicies)
	***/
		WHEN	((#InfRi_Total_Initial.SumBaseCcyAmtRoeDateBooked <=  0.0001)  	AND
			 (#InfRi_Total_Initial.SumBaseCcyAmtRoeDateBooked >= -0.0001))		

			THEN (1.00/#InfRi_Total.NbrRiPolicies)

		ELSE	(#InfRi.SumBaseCcyAmtRoeDateBooked / #InfRi_Total_Initial.SumBaseCcyAmtRoeDateBooked)
	END	AS RiRatio

INTO	#InfRi_Ratio

FROM	#InfRi

LEFT JOIN #InfRi_Total ON
	#InfRi.FKGenIntStatsHeader = #InfRi_Total.FKGenIntStatsHeader	AND
	#InfRi.FKGenIntStatsSec = #InfRi_Total.FKGenIntStatsSec		AND
        #InfRi.FKGenIntStatsAmt = #InfRi.FKGenIntStatsAmt              AND
	#InfRi.FKRevenueCode = #InfRi_Total.FKRevenueCode

LEFT JOIN #InfRi_Total_Initial ON
	#InfRi.FKGenIntStatsHeader = #InfRi_Total_Initial.FKGenIntStatsHeader	AND
	#InfRi.FKGenIntStatsSec = #InfRi_Total_Initial.FKGenIntStatsSec		AND
        #InfRi.FKGenIntStatsAmt = #InfRi_Total_Initial.FKGenIntStatsAmt         AND
	#InfRi.FKRevenueCode = #InfRi_Total_Initial.FKRevenueCode

CREATE INDEX #InfRi_Ratio_PrimaryKey ON #InfRi_Ratio ([FKGenRiIntPolNbr], [FKGenRiIntPolSeq])

	(SELECT 
		AllTransFullDetail.FKGenIntStatsHeader,
		AllTransFullDetail.FKGenIntStatsSec,
		AllTransFullDetail.FKGenIntStatsAmt
	 FROM 	AllTransFullDetail

	 WHERE
		AllTransFullDetail.PFKRiPolNbr <> '' 		AND 
		AllTransFullDetail.PFKRiPolNbr IS NOT NULL
	
	 GROUP BY
		AllTransFullDetail.FKGenIntStatsHeader,
		AllTransFullDetail.FKGenIntStatsSec,
                AllTransFullDetail.FKGenIntStatsAmt

	 )AS RIBrokerageTransactions on
		RIBrokerageTransactions.FKGenIntStatsHeader = AllTransFullDetail.FKGenIntStatsHeader		AND
		RIBrokerageTransactions.FKGenIntStatsSec  = AllTransFullDetail.FKGenIntStatsSec AND
                RIBrokerageTransactions.FKGenIntStatsAmt  = AllTransFullDetail.FKGenIntStatsAmt

/***	Gross Transaction Records with Brokerage which have a matching Ri Transaction		***/
WHERE 
	(AllTransFullDetail.PFKRiPolNbr = '' OR AllTransFullDetail.PFKRiPolNbr IS NULL)		AND
	 AllTransFullDetail.FKRevenueCode = 'BRK'						AND
	 RIBrokerageTransactions.FKGenIntStatsHeader IS NOT NULL


GROUP BY
	AllTransFullDetail.FKGenIntStatsHeader,
	AllTransFullDetail.FKGenIntStatsSec,
        AllTransFullDetail.FKGenIntStatsAmt

	 	FKGenIntStatsHeader,

		GrossBrokerageTransactions.FKGenIntStatsSec = AllTransFullDetail.FKGenIntStatsSec AND

	/***	Ri Brokerage Transactions		***/
	AllTransFullDetail.PFKRiPolNbr <> ''			AND
	AllTransFullDetail.PFKRiPolNbr IS NOT NULL		AND
	AllTransFullDetail.FKRevenueCode = 'BRK' 		AND
